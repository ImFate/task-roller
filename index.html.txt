<script>
  const SAMPLE_TASKS = [
    { id: 1, points: 950, boss: "Abyssal Sire", desc: "Create a Bludgeon (3 Unsireds completes the task)" },
    { id: 2, points: 315, boss: "Abyssal Sire", desc: "Obtain 1 Unsired" },
    { id: 3, points: 2100, boss: "Abyssal Sire", desc: "Obtain an Abyssal Dagger (Abby Demons is okay)" },
    { id: 4, points: 75, boss: "Abyssal Sire", desc: "3 unique people complete the CA: Perfect Sire" },
    { id: 5, points: 4500, boss: "Alchemical Hydra", desc: "Obtain a Hydra Claw" },
    { id: 6, points: 5200, boss: "Alchemical Hydra", desc: "Create a Dragon Hunter Lance (Claw + Zammy Spear)" },
    { id: 7, points: 2400, boss: "Alchemical Hydra", desc: "Create a Brimstone Ring" },
    { id: 8, points: 100, boss: "Alchemical Hydra", desc: "2 unique people complete the CA: No Pressure" },
    { id: 9, points: 1000, boss: "Alchemical Hydra", desc: "Fill 3 Hydra collection log slots (dupes count)" },
  ];

  const STORAGE_KEY = "perkWars_taskRoller_v2";
  const STORAGE_FILTER_KEY = "perkWars_taskRoller_filters_v2";
  const STORAGE_ROLLCOUNT_KEY = "perkWars_taskRoller_rollcount_v2";

  let tasks = [];
  let enabledBosses = new Set();
  let lastRolledIndices = []; // store last roll set

  const $ = (sel) => document.querySelector(sel);

  function safeParseNumber(x){
    const n = Number(String(x).replace(/[^\d.-]/g,""));
    return Number.isFinite(n) ? n : 0;
  }

  function uniqBosses(list){
    const set = new Set(list.map(t => (t.boss || "").trim()).filter(Boolean));
    return Array.from(set).sort((a,b)=>a.localeCompare(b));
  }

  function saveState(){
    localStorage.setItem(STORAGE_KEY, JSON.stringify(tasks));
    localStorage.setItem(STORAGE_FILTER_KEY, JSON.stringify(Array.from(enabledBosses)));
    localStorage.setItem(STORAGE_ROLLCOUNT_KEY, $("#rollCount").value);
    $("#storageInfo").textContent = "Saved in browser storage";
  }

  function loadState(){
    const saved = localStorage.getItem(STORAGE_KEY);
    const savedFilters = localStorage.getItem(STORAGE_FILTER_KEY);
    const savedRollCount = localStorage.getItem(STORAGE_ROLLCOUNT_KEY);

    if (saved){
      try{
        const parsed = JSON.parse(saved);
        tasks = (Array.isArray(parsed) && parsed.length) ? parsed : [...SAMPLE_TASKS];
      } catch {
        tasks = [...SAMPLE_TASKS];
      }
    } else {
      tasks = [...SAMPLE_TASKS];
    }

    const bossesNow = uniqBosses(tasks);

    if (savedFilters){
      try{
        const parsed = JSON.parse(savedFilters);
        enabledBosses = new Set(Array.isArray(parsed) ? parsed : bossesNow);
      } catch {
        enabledBosses = new Set(bossesNow);
      }
    } else {
      enabledBosses = new Set(bossesNow);
    }

    // prune any old bosses that no longer exist
    const bossesSet = new Set(bossesNow);
    enabledBosses = new Set(Array.from(enabledBosses).filter(b => bossesSet.has(b)));

    if (savedRollCount && ["1","2","3"].includes(savedRollCount)){
      $("#rollCount").value = savedRollCount;
    }

    refreshUI();
  }

  function parseTasksFromText(text){
    const lines = text.split(/\r?\n/).map(l => l.trim()).filter(l => l.length && !l.startsWith("#"));
    const parsed = [];

    for (const line of lines){
      let parts;
      if (line.includes("|")) parts = line.split("|").map(s => s.trim());
      else if (line.includes("\t")) parts = line.split("\t").map(s => s.trim());
      else parts = line.split(",").map(s => s.trim());

      if (parts.length < 4) continue;

      const id = parts[0];
      const points = safeParseNumber(parts[1]);
      const boss = parts[2];
      const desc = parts.slice(3).join(" | ");

      if (!boss || !desc) continue;

      parsed.push({ id: isNaN(Number(id)) ? id : Number(id), points, boss: boss.trim(), desc: desc.trim() });
    }
    return parsed;
  }

  function exportTasksToText(){
    return tasks.map(t => `${t.id} | ${t.points} | ${t.boss} | ${t.desc}`).join("\n");
  }

  function getEligibleTaskIndices(){
    const eligible = [];
    for (let i=0; i<tasks.length; i++){
      if (enabledBosses.has(tasks[i].boss)) eligible.push(i);
    }
    return eligible;
  }

  // pick N unique indices from eligible
  function sampleUnique(eligible, n){
    // Fisher-Yates shuffle copy, then take first n
    const arr = eligible.slice();
    for (let i = arr.length - 1; i > 0; i--){
      const j = Math.floor(Math.random() * (i + 1));
      [arr[i], arr[j]] = [arr[j], arr[i]];
    }
    return arr.slice(0, n);
  }

  function rollTasks(avoidSameSet=false){
    const eligible = getEligibleTaskIndices();
    if (!eligible.length){
      showMessage("No tasks available. Enable at least one boss/raid (or load tasks).", true);
      return;
    }

    const wanted = Number($("#rollCount").value);
    const n = Math.min(wanted, eligible.length);

    let picked = sampleUnique(eligible, n);

    // avoid repeating the exact same set as last time (if possible)
    if (avoidSameSet && eligible.length > 1 && lastRolledIndices.length){
      const lastKey = lastRolledIndices.slice().sort((a,b)=>a-b).join(",");
      for (let tries=0; tries<12; tries++){
        const candidate = sampleUnique(eligible, n);
        const key = candidate.slice().sort((a,b)=>a-b).join(",");
        if (key !== lastKey){
          picked = candidate;
          break;
        }
      }
    }

    lastRolledIndices = picked.slice();
    renderResults(picked.map(i => tasks[i]));
    renderStatus();
  }

  function renderResults(taskList){
    const count = taskList.length;
    const totalPoints = taskList.reduce((sum,t)=>sum + (Number(t.points)||0), 0);

    const blocks = taskList.map((t, idx) => `
      <div class="taskCard">
        <div class="taskHead">
          <div style="font-weight:900">Task ${idx+1}</div>
          <span class="tag">${escapeHtml(t.boss)}</span>
          <span class="spacer"></span>
          <span class="tiny">#${escapeHtml(String(t.id))}</span>
        </div>
        <div class="kv">
          <div class="k">Task #</div><div class="v">${escapeHtml(String(t.id))}</div>
          <div class="k">Points</div><div class="v">${escapeHtml(String(t.points))}</div>
          <div class="k">Boss</div><div class="v">${escapeHtml(t.boss)}</div>
        </div>
        <div class="desc">${escapeHtml(t.desc)}</div>
      </div>
    `).join("");

    $("#resultBox").innerHTML = `
      <div class="row">
        <div style="font-weight:900; font-size:16px">Rolled ${count} Task${count===1?"":"s"}</div>
        <span class="pill">Total points: ${escapeHtml(String(totalPoints))}</span>
        <span class="spacer"></span>
        <span class="tiny">No duplicates within the roll</span>
      </div>
      <div class="resultGrid">${blocks}</div>
    `;
  }

  function renderBossFilters(){
    const bosses = uniqBosses(tasks);
    const container = $("#bossList");
    container.innerHTML = "";

    bosses.forEach(boss => {
      const id = "boss_" + boss.replace(/[^a-z0-9]/gi, "_");
      const checked = enabledBosses.has(boss);

      const el = document.createElement("label");
      el.className = "bossItem";
      el.setAttribute("for", id);
      el.innerHTML = `
        <input type="checkbox" id="${id}" ${checked ? "checked" : ""} />
        <span>${escapeHtml(boss)}</span>
        <small>${checked ? "enabled" : "disabled"}</small>
      `;
      container.appendChild(el);

      const checkbox = el.querySelector("input");
      checkbox.addEventListener("change", () => {
        if (checkbox.checked) enabledBosses.add(boss);
        else enabledBosses.delete(boss);
        el.querySelector("small").textContent = checkbox.checked ? "enabled" : "disabled";
        saveState();
        renderStatus();
      });
    });
  }

  function renderTaskTable(){
    const body = $("#taskTableBody");
    body.innerHTML = "";
    tasks.forEach(t => {
      const tr = document.createElement("tr");
      tr.innerHTML = `
        <td>${escapeHtml(String(t.id))}</td>
        <td>${escapeHtml(String(t.points))}</td>
        <td>${escapeHtml(t.boss)}</td>
        <td>${escapeHtml(t.desc)}</td>
      `;
      body.appendChild(tr);
    });
  }

  function renderStatus(){
    const eligibleCount = getEligibleTaskIndices().length;
    const total = tasks.length;
    const bossesTotal = uniqBosses(tasks).length;
    const bossesEnabled = Array.from(enabledBosses).filter(b => uniqBosses(tasks).includes(b)).length;

    const bar = $("#statusBar");
    const wanted = Number($("#rollCount").value);
    const effective = Math.min(wanted, eligibleCount);

    bar.innerHTML = `
      <span class="badge ${eligibleCount ? "good" : "bad"}">Eligible tasks: <b>${eligibleCount}</b> / ${total}</span>
      <span class="badge">Bosses enabled: <b>${bossesEnabled}</b> / ${bossesTotal}</span>
      <span class="badge">Roll size: <b>${effective}</b> (requested ${wanted})</span>
    `;
  }

  function showMessage(msg, isError=false){
    $("#resultBox").innerHTML = `<div class="badge ${isError ? "bad" : "good"}">${escapeHtml(msg)}</div>`;
    renderStatus();
  }

  function escapeHtml(str){
    return String(str).replace(/[&<>"']/g, (m) => ({
      "&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;","'":"&#039;"
    }[m]));
  }

  function refreshUI(){
    renderBossFilters();
    renderTaskTable();
    $("#taskInput").value = exportTasksToText();
    renderStatus();
    saveState();
  }

  // Events
  $("#rollBtn").addEventListener("click", () => rollTasks(false));
  $("#rerollBtn").addEventListener("click", () => rollTasks(true));

  $("#rollCount").addEventListener("change", () => {
    saveState();
    renderStatus();
  });

  $("#loadBtn").addEventListener("click", () => {
    const parsed = parseTasksFromText($("#taskInput").value);
    if (!parsed.length){
      showMessage("Could not parse tasks. Each line needs 4 fields: # | points | boss | description", true);
      return;
    }
    tasks = parsed;

    // keep filter prefs where possible; enable any new bosses by default
    const bosses = uniqBosses(tasks);
    const oldEnabled = new Set(enabledBosses);
    enabledBosses = new Set();
    bosses.forEach(b => enabledBosses.add(oldEnabled.has(b) ? b : b));

    lastRolledIndices = [];
    refreshUI();
    showMessage("Loaded tasks successfully.");
  });

  $("#exportBtn").addEventListener("click", async () => {
    const text = exportTasksToText();
    try{
      await navigator.clipboard.writeText(text);
      showMessage("Export copied to clipboard.");
    } catch {
      const ta = $("#taskInput");
      ta.value = text;
      ta.focus();
      ta.select();
      showMessage("Export ready (clipboard blocked). Select/copy from the textarea.");
    }
  });

  $("#enableAllBtn").addEventListener("click", () => {
    uniqBosses(tasks).forEach(b => enabledBosses.add(b));
    refreshUI();
  });

  $("#disableAllBtn").addEventListener("click", () => {
    enabledBosses.clear();
    refreshUI();
  });

  $("#resetBtn").addEventListener("click", () => {
    tasks = [...SAMPLE_TASKS];
    enabledBosses = new Set(uniqBosses(tasks));
    lastRolledIndices = [];
    $("#rollCount").value = "1";
    refreshUI();
    showMessage("Reset to sample tasks.");
  });

  // Boot
  $("#storageInfo").textContent = "Saved in browser storage";
  loadState();
</script>
</body>
